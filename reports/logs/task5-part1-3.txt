CREATE TABLE shop.orders (
    customer text,
    order_date timestamp,
    order_id uuid,
    delivery text,
    total decimal,
    item_ids list<uuid>,
    PRIMARY KEY (customer, order_date, order_id)
) WITH CLUSTERING ORDER BY (order_date DESC, order_id ASC)
    AND additional_write_policy = '99p'
