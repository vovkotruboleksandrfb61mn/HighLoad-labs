# За назвою в межах категорії, через view items_by_name:
cqlsh:shop> SELECT category, name, producer, price FROM items_by_name WHERE category = 'Ноутбуки' AND name = 'ThinkPad T14';
 category | name         | producer | price
----------+--------------+----------+-------
 Ноутбуки | ThinkPad T14 |   Lenovo | 47999

(1 rows)

# Ціна в проміжку в межах категорії: зріз за clustering column price:
cqlsh:shop> SELECT name, producer, price FROM items WHERE category = 'Ноутбуки' AND price >= 35000 AND price <= 50000;
 name            | producer | price
-----------------+----------+-------
 Zenbook 14 OLED |     ASUS | 39999
    ThinkPad T14 |   Lenovo | 47999

(2 rows)

# Ціна та виробник, через view items_by_producer:
cqlsh:shop> SELECT name, producer, price FROM items_by_producer WHERE category = 'Телефони' AND producer = 'Samsung' AND price < 15000;
 name       | producer | price
------------+----------+-------
 Galaxy A55 |  Samsung | 12999

(1 rows)

# За вмістом map, через індекси на props (пари ключ-значення і ключі):

 name          | price | props
---------------+-------+-----------------------------------------------------------

(2 rows)
